# ============================================================
# MAITRI ANTARCTIC RESEARCH STATION
# SPECIFICATION DEFINITIONS
# ============================================================
#
# This file describes engineering and semantic specifications.
#
# Syntax:
#
# <name>:<type> {
#       property=value
# }
#
# Type defaults:
#
# default:<type> {
#       property=value
# }
#
# A component inherits from default:<type>.
# Component properties override inherited properties.
#
# Ranges:
#
# rating=-50:50 unit=C
#
# Single measurements:
#
# power=300 unit=kW
#
# Strings:
#
# provides="electrical power"
#
# Visual/physical properties are optional and are NOT required
# for every component.
#
# ============================================================
